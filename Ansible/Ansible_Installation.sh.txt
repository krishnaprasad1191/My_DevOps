#!/bin/bash

uid=$(id -u)

val(){

        if [ $1 -ne 0 ];then
                echo " $2 failed"
        else
                echo " $2 Succeed"
        fi
}

if [ $uid -eq 0 ]; then

        apt install -y python3
        val $? "Python"
        apt install -y python3-pip
        val $? "pip"
        apt install ansible
        val $? "Ansible Installation.."
else
        echo "Must be root user"
        exit 1
fi
